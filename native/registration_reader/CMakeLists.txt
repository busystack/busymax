cmake_minimum_required(VERSION 3.14)
project(busymax_registration_reader LANGUAGES CXX)
add_library(busymax_registration_reader SHARED registration_reader.cc)
target_compile_features(busymax_registration_reader PRIVATE cxx_std_17)
set_target_properties(busymax_registration_reader PROPERTIES CXX_VISIBILITY_PRESET hidden)
if(MSVC)
  target_compile_options(busymax_registration_reader PRIVATE /W4 /WX)
else()
  target_compile_options(busymax_registration_reader PRIVATE -Wall -Wextra -Werror)
endif()
add_executable(busymax_registration_reader_tests EXCLUDE_FROM_ALL
  registration_reader_test.cc registration_reader.cc)
target_compile_features(busymax_registration_reader_tests PRIVATE cxx_std_17)
target_compile_definitions(busymax_registration_reader_tests PRIVATE BUSYMAX_READER_TESTING)
enable_testing()
add_test(NAME busymax_registration_reader_tests COMMAND busymax_registration_reader_tests)
